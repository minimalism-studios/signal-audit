<script>
(function () {
  function fixFaqFooter() {
    const faq = document.querySelector('.signal-faq-page');
    const footer = document.querySelector('#footer-sections');

    if (!faq || !footer) return;

    footer.style.marginTop = '0px';

    const faqBottom = faq.getBoundingClientRect().bottom + window.scrollY;
    const footerTop = footer.getBoundingClientRect().top + window.scrollY;
    const overlap = faqBottom - footerTop;

    const buffer = window.innerWidth <= 767 ? 620 : 48;

    if (overlap > 0) {
      footer.style.marginTop = Math.ceil(overlap + buffer) + 'px';
    }
  }

  window.addEventListener('load', fixFaqFooter);
  window.addEventListener('resize', fixFaqFooter);
  setTimeout(fixFaqFooter, 300);
  setTimeout(fixFaqFooter, 1000);
  setTimeout(fixFaqFooter, 1800);
})();
</script>
