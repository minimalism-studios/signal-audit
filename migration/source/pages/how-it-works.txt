(function () {

  const newSrc = "https://images.squarespace-cdn.com/content/698f8549821d5e2aaf4d4edc/6fb268da-2e08-4e25-a40a-e4d03f62c8a9/web-logo-white.png?format=1500w";

  function swapLogo() {

    const logos = document.querySelectorAll(
      ".header-display-desktop .header-title-logo img, .header-display-mobile .header-title-logo img"
    );

    if (!logos.length) return;

    logos.forEach((logo) => {

      logo.src = newSrc;
      logo.srcset = newSrc;

      logo.setAttribute("src", newSrc);
      logo.setAttribute("srcset", newSrc);

    });
  }

  document.addEventListener("DOMContentLoaded", swapLogo);
  window.addEventListener("load", swapLogo);

  setTimeout(swapLogo, 500);
  setTimeout(swapLogo, 1500);
  setTimeout(swapLogo, 3000);

})();
