            <summary>
              Is there a fee for the 30-day evaluation?
              <span aria-hidden="true">+</span>
            </summary>

            <div class="vanguard-faq__answer">
              <p>
                No. Selected Vanguard organizations pay no evaluation fee. The
                standard deployment value is $5,000, but participation requires
                active technical involvement, weekly feedback, and a final
                evaluation meeting.
              </p>
            </div>
          </details>

          <details class="vanguard-faq__item">
            <summary>
              Do we have to purchase Signal Audit afterward?
              <span aria-hidden="true">+</span>
            </summary>

            <div class="vanguard-faq__answer">
              <p>
                No. The organization may continue, extend for a specific
                technical reason, or conclude the evaluation. There is no
                obligation to purchase.
              </p>
            </div>
          </details>

          <details class="vanguard-faq__item">
            <summary>
              Can we evaluate only one observability integration?
              <span aria-hidden="true">+</span>
            </summary>

            <div class="vanguard-faq__answer">
              <p>
                Yes. Organizations may evaluate Signal Audit with Grafana,
                Datadog, or both, depending on their current environment and
                evaluation goals.
              </p>
            </div>
          </details>

          <details class="vanguard-faq__item">
            <summary>
              How much engineering effort is required?
              <span aria-hidden="true">+</span>
            </summary>

            <div class="vanguard-faq__answer">
              <p>
                The primary technical work involves assigning a point of
                contact, configuring a webhook connection, validating test
                events, and attending brief weekly reviews. The final effort
                depends on the organization’s security and approval process.
              </p>
            </div>
          </details>

          <details class="vanguard-faq__item">
            <summary>
              Can we stop the evaluation early?
              <span aria-hidden="true">+</span>
            </summary>

            <div class="vanguard-faq__answer">
              <p>

                agreed pilot terms. Integrations will be deactivated and data
                will be handled according to the evaluation agreement.

<script>
(function () {
  const VIDEO_ID = 'GMix56shEtk';

  function trackVanguardDemoEvent(eventName, params = {}) {
    const payload = {
      page_path: window.location.pathname,
      page_location: window.location.href,
      video_id: VIDEO_ID,
      video_title: 'Signal Audit Demo | From Production Telemetry to Operational Intelligence',
      ...params
    };

    if (typeof window.gtag === 'function') {
      window.gtag('event', eventName, payload);
      return;
    }

    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({
      event: eventName,
      ...payload
    });
  }

  function initVanguardDemoTracking() {
    const section =
      document.getElementById('vanguard-demo');

    const playerElement =
      document.getElementById('vanguard-demo-player');

    const applicationLink =
      document.querySelector('[data-vanguard-demo-apply]');

    /* -------------------------------------
       DEMO IMPRESSION
    ------------------------------------- */

    if (
      section &&
      'IntersectionObserver' in window
    ) {
      let impressionTracked = false;

      const observer = new IntersectionObserver(
        (entries) => {
          entries.forEach((entry) => {
            if (
              entry.isIntersecting &&
              !impressionTracked
            ) {
              impressionTracked = true;

              trackVanguardDemoEvent(
                'vanguard_demo_impression',
                {
                  percent_visible: 40
                }
              );

              observer.disconnect();
            }
          });
        },
        {
          threshold: 0.4
        }
      );

      observer.observe(section);
    }

    /* -------------------------------------
       APPLICATION CTA
    ------------------------------------- */

    if (applicationLink) {
      applicationLink.addEventListener(
        'click',
        () => {
          trackVanguardDemoEvent(
            'vanguard_demo_application_click',
            {
              link_url: applicationLink.href,
              link_text: 'Apply for the Program'
            }
          );
        }
      );
    }

    /* -------------------------------------
       YOUTUBE PLAY
    ------------------------------------- */

    if (!playerElement) return;

    let playTracked = false;

    function createPlayer() {
      if (
        !window.YT ||
        !window.YT.Player
      ) {
        return;
      }

      new window.YT.Player(
        'vanguard-demo-player',
        {
          events: {
            onStateChange(event) {
              if (
                event.data ===
                  window.YT.PlayerState.PLAYING &&
                !playTracked
              ) {
                playTracked = true;

                trackVanguardDemoEvent(
                  'vanguard_demo_play',
                  {
                    video_provider: 'youtube'
                  }
                );
              }
            }
          }
        }
      );
    }

    if (
      window.YT &&
      window.YT.Player
    ) {
      createPlayer();
      return;
    }

    const previousCallback =
      window.onYouTubeIframeAPIReady;

    window.onYouTubeIframeAPIReady =
      function () {
        if (
          typeof previousCallback ===
          'function'
        ) {
          previousCallback();
        }

        createPlayer();
      };

    if (
      !document.querySelector(
        'script[src="https://www.youtube.com/iframe_api"]'
      )
    ) {
      const script =
        document.createElement('script');

      script.src =
        'https://www.youtube.com/iframe_api';

      document.head.appendChild(script);
    }
  }

  if (
    document.readyState === 'loading'
  ) {
    document.addEventListener(
      'DOMContentLoaded',
      initVanguardDemoTracking
    );
  } else {
    initVanguardDemoTracking();
  }
})();
</script>
