<script>
(function () {

  function mountSecurityPage() {
    const body = document.body;

    if (
      !body.classList.contains(
        'collection-6a9fb3323d40f31a3f11e1e1'
      )
    ) {
      return;
    }

    const main = document.getElementById('page');

    const securityPage =
      document.querySelector('.sa-security-page');

    if (!main || !securityPage) {
      return;
    }

    /*
     * Prevent running twice.
     */
    if (
      main.classList.contains('sa-security-host')
    ) {
      return;
    }

    /*
     * The Security page CSS currently lives inside the
     * Squarespace Code Block that we are about to remove.
     *
     * Move those styles into <head> first so they remain
     * active after the Fluid Engine wrapper is removed.
     */
    const codeContainer =
      securityPage.closest('.sqs-code-container');

    if (codeContainer) {

      codeContainer
        .querySelectorAll('style')
        .forEach(function (style) {

          document.head.appendChild(style);

        });

    }

    /*
     * Remove Squarespace Fluid Engine from the equation.
     *
     * This moves the actual Security page directly beneath
     * <main id="page"> so its natural content height controls
     * the document.
     */
    main.replaceChildren(securityPage);

    main.classList.add(
      'sa-security-host'
    );

  }


  if (
    document.readyState === 'loading'
  ) {

    document.addEventListener(
      'DOMContentLoaded',
      mountSecurityPage,
      { once: true }
    );

  } else {

    mountSecurityPage();

  }

})();
</script>
